Google Sheets RSVP Integration

Paste your deployed Google Apps Script Web App URL below so every RSVP submission automatically updates Sheet 1 (Guest Database) and Sheet 2 (RSVP Responses). 

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = ss.getSheetByName("Guest Database");
  var respSheet = ss.getSheetByName("RSVP Responses");
  var data = JSON.parse(e.postData.contents);

  respSheet.appendRow([
    data.mainGuest,
    data.attendance,
    data.companions,
    data.totalAttendees,
    new Date()
  ]);

  var rows = dbSheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0] === data.mainGuest) {
      dbSheet.getRange(i + 1, 7).setValue(data.attendance);
      break;
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}
